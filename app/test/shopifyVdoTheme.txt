      <!-- Button below video -->
      <button
        onclick="openChatbot()"
        style="max-width: 110px; background: #fb90a2; color: rgb(var(--color-button-text)); font-size: 12px; text-align: center; border-radius: 6px; font-family: var(--font-body-family); border: none; outline: none; cursor: pointer;"
        class="video-avatar-button md:max-w-32 md:p-2 md:text-xs md:rounded-full"
      >

    <!-- Chatbot iframe container -->
    <div
      id="chatbot-iframe-container"
      style="display: none; position: fixed; top: 0; left: 0; width: 100%; height: 100vh; z-index: 1000; background: rgb(var(--color-background));"
    >
      <iframe
        id="chatbot-iframe"
        src="https://pov-shopify.vercel.app/"
        frameborder="0"
        style="width: 100%; height: 100%;"
        allow="microphone"
      ></iframe>
    </div>

    <script>
      // capture current page
      const currentPage = window.location.pathname.split('/').filter(Boolean).pop() || 'unknown-page';

      function openChatbot() {
        const iframe = document.getElementById('chatbot-iframe');
        document.getElementById('chatbot-iframe-container').style.display = 'block';
        document.getElementById('chatbot-iframe-container').classList.add('active');
        document.body.classList.add('iframe-open');
        document.querySelector('.video-avatar-container').style.display = 'none';

        // Send the current page name to the iframe
        iframe.contentWindow.postMessage({
          type: 'PAGE_INFO',
          payload: {
            pageName: currentPage, // This will be "new-collection" on the new-collection page
            fullPath: window.location.pathname
          }
        }, 'https://pov-shopify.vercel.app/');
      }

      function closeChatbot() {
        document.getElementById('chatbot-iframe-container').style.display = 'none';
        document.getElementById('chatbot-iframe-container').classList.remove('active');
        document.body.classList.remove('iframe-open');
        document.querySelector('.video-avatar-container').style.display = 'flex';
      }

      // Combined message handler for both chatbot and cart functionality
      window.addEventListener('message', (event) => {
        // Verify the origin
        if (event.origin !== 'https://pov-shopify.vercel.app/') return;

        const { type, payload } = event.data || {};

        // Handle chatbot close action
        if (event.data.action === 'closeChatbot') {
          closeChatbot();
        }

        // Handle add to cart
        if (type === 'ADD_TO_CART' && payload?.variantId && payload?.quantity) {
          fetch('/cart/add.js', {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
              Accept: 'application/json',
            },
            body: JSON.stringify({
              id: payload.variantId,
              quantity: payload.quantity
            }),
          })
            .then((res) => {
              if (!res.ok) throw new Error('Failed to add to cart');
              return res.json();
            })
            .then(() => fetch('/cart.js'))
            .then((res) => res.json())
            .then((cart) => {
              // Send updated cart info back to iframe
              event.source.postMessage({
                type: 'CART_INFO',
                payload: {
                  itemCount: cart.item_count,
                }
              }, event.origin);
            })
            .catch((error) => {
              console.error('Cart error:', error);
              event.source.postMessage({
                type: 'CART_ERROR',
                payload: { error: 'Failed to add item to cart' }
              }, event.origin);
            });
        }

        // Handle get cart item count
        if (type === 'GET_CART') {
          fetch('/cart.js')
            .then((res) => res.json())
            .then((cart) => {
              event.source.postMessage({
                type: 'CART_INFO',
                payload: {
                  itemCount: cart.item_count,
                }
              }, event.origin);
            })
            .catch((error) => {
              console.error('Cart fetch error:', error);
              event.source.postMessage({
                type: 'CART_ERROR',
                payload: { error: 'Failed to fetch cart' }
              }, event.origin);
            });
        }
      });

      // Send theme data to iframe
      window.addEventListener('load', () => {
        const iframe = document.getElementById('chatbot-iframe');
        iframe.contentWindow.postMessage({
          theme: {
            primaryColor: 'rgb(var(--color-button))',
            font: 'var(--font-body-family)',
            background: 'rgb(var(--color-primary))',
            buttonText: 'rgb(var(--color-button-text))'
          }
        }, 'https://pov-shopify.vercel.app/');
      });

      // Add media query class for responsive styling
      function applyResponsiveStyles() {
        const button = document.querySelector('.video-avatar-button');
        if (window.matchMedia('(min-width: 768px)').matches) {
          button.classList.add('md:max-w-32', 'md:p-2', 'md:text-xs', 'md:rounded-full');
        } else {
          button.classList.remove('md:max-w-32', 'md:p-2', 'md:text-xs', 'md:rounded-full');
        }
      }

      window.addEventListener('load', applyResponsiveStyles);
      window.addEventListener('resize', applyResponsiveStyles);
    </script>
